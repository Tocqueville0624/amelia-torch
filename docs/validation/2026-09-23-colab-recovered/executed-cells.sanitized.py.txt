# Historical cell 2 (sanitized, not a replay recipe)
import datetime, json, time
assert setup.returncode == 0, "Fix and retain setup failure before validation"
PYTHON = str(REPO / ".venv/bin/python")
OUT = REPO / "results/local/cloud"
steps = []
def checked_step(name, command):
    print("START", name, flush=True)
    started = time.monotonic()
    result = subprocess.run(command, cwd=REPO, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
    text = result.stdout.replace(str(REPO), "<project>").replace(str(pathlib.Path.home()), "<home>")
    (OUT / (name + ".log")).write_text(text)
    steps.append({"name": name, "exit_status": result.returncode, "seconds": time.monotonic() - started})
    (OUT / "validation-steps.json").write_text(json.dumps({"commit": REVISION, "updated_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(), "steps": steps}, indent=2))
    print("END", name, result.returncode, text[-2200:], flush=True)
    return result.returncode
checked_step("cuda-probe", [PYTHON, "-m", "amelia_torch.diagnostics", "--require-device", "cuda", "--output", str(OUT / "cuda-probe.json")])
checked_step("pytest", [PYTHON, "-m", "pytest", "-q"])
checked_step("ruff", [PYTHON, "-m", "ruff", "check", "src", "tests", "scripts"])
for case in ("bridge", "compatibility", "torch-compat", "reference_metadata", "downstream"):
    checked_step("r-test-" + case, ["Rscript", "r-package/tests/" + case + ".R"])
for dtype in ("float64", "float32"):
    checked_step("native-" + dtype, [PYTHON, "scripts/validate_accelerator.py", "--device", "cuda", "--dtype", dtype, "--output", str(OUT / ("native-" + dtype + "-validation.json"))])
    checked_step("r-" + dtype, ["Rscript", "scripts/validate_r_accelerator.R", "cuda", dtype, str(OUT / ("r-" + dtype + "-validation.json"))])
assert all(step["exit_status"] == 0 for step in steps), "Validation failures retained; inspect before benchmarking"
print("ALL_CLOUD_VALIDATION_STEPS_PASSED", len(steps))
